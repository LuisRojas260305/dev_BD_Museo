# ------------------------------------------------------------
# 5. INSTALACION RAPIDA
# ------------------------------------------------------------
#
#  1. Instalar todo el software de las secciones 1, 2 y 3
#  2. Crear la base MySQL:
#       mysql -u root < Backend/scripts/sql/schema.sql
#       mysql -u root < Backend/scripts/sql/seed.sql
#  3. Instalar dependencias de cada servicio:
#       cd Backend              && npm install
#       cd Backend/mongodb-service   && npm install
#       cd Backend/cassandra-service && npm install
#  4. Crear las tablas de Cassandra:
#       cd Backend/cassandra-service && node init-db.js
#  5. Arrancar todo con el script del escritorio:
#       iniciar_museo.bat
#
#  El seed de MongoDB se ejecuta automaticamente si la BD esta vacia.
# ------------------------------------------------------------
